let score = 0; let level = 1; let highScore = localStorage.getItem("highScore") || 0; document.getElementById("highScore").innerText = highScore; function startGame() { document.getElementById("menu").style.display = "none"; document.getElementById("game").style.display = "block"; createGrid(); } function createGrid() { let grid = document.getElementById("grid"); grid.innerHTML = ""; let random = Math.floor(Math.random() * 9); for (let i = 0; i < 9; i++) { let b = document.createElement("div"); b.className = "box"; if (i === random) { b.classList.add("diff"); b.onclick = function () { score += 10; if (score % 50 === 0) { level++; } document.getElementById("score").innerText = score; document.getElementById("level").innerText = level; if (score > highScore) { highScore = score; localStorage.setItem("highScore", highScore); document.getElementById("highScore").innerText = highScore; } createGrid(); }; } else { b.onclick = function () { alert("Wrong! Try Again"); }; } grid.appendChild(b); } } function restartGame() { score = 0; level = 1; document.getElementById("score").innerText = score; document.getElementById("level").innerText = level; createGrid(); }
